chiara-api-example.js 
// Fetch live sensor readings
const response = await fetch(
  'https://api.chiaraos.com/v1/buildings/BLD-001/sensors',
  {
    headers: {
      'Authorization': `Bearer ${API_KEY}`,
      'Content-Type': 'application/json'
    }
  }
);

const { sensors } = await response.json();

// Returns real-time data for all sensors
sensors.forEach(s => {
  console.log(`${s.name}: ${s.value} ${s.unit}`);
});
